/**
 * A measuring probe for row 17z16b of effect-slot-refactor.md, not a test: copy it into the application module's
 * {@code service/trigger} test package as {@code Run17z16ProbeTest.java} and run it. For each trigger event it puts a
 * fixture ability with a plain effect beside the same ability with a "you may", a target, two target groups, a
 * graveyard target, a mode chosen as it triggers and such a mode with a target, answers the choices and writes to
 * {@link #OUT} which fields of the event's record differ between the two stack entries.
 */

    private static Card card(String name, CardType type, CardSubtype... subtypes) {
        Card card = new Card();
        card.setName(name);
        card.setType(type);
        card.setSubtypes(List.of(subtypes));
        card.setManaCost("{1}");
        if (type == CardType.CREATURE) {
            card.setPower(2);
            card.setToughness(2);
        }
        return card;
    }

    private static ChooseOneAtTriggerTimeEffect gainOrDraw() {
        return new ChooseOneAtTriggerTimeEffect(new ChooseOneEffect(List.of(
                new ChooseOneEffect.ChooseOneOption("You gain 1 life", new GainLifeEffect(1)),
                new ChooseOneEffect.ChooseOneOption("Draw a card", new DrawCardEffect(1)))));
    }

    /** The variants of one ability on {@code trigger}: name suffix to card. */
    private Map<String, Card> variants(String prefix, CardType type, Trigger trigger, CardSubtype... subtypes) {
        Map<String, Card> cards = new LinkedHashMap<>();
        Card plain = card(prefix + " plain", type, subtypes);
        plain.addTrigger(trigger, new GainLifeEffect(1));
        cards.put("plain", plain);

        Card may = card(prefix + " may", type, subtypes);
        may.addTrigger(trigger, new MayEffect(new GainLifeEffect(1), "Gain?"));
        cards.put("may", may);

        Card target = card(prefix + " target", type, subtypes);
        CardEffect boost = new BoostTargetCreatureEffect(1, 1);
        target.addTrigger(trigger, boost);
        target.target(TargetFilters.creature()).bindEffect(boost);
        cards.put("target", target);

        Card two = card(prefix + " twogroups", type, subtypes);
        CardEffect destroy = new DestroyTargetPermanentEffect();
        CardEffect boost2 = new BoostTargetCreatureEffect(1, 1);
        two.addTrigger(trigger, destroy, boost2);
        two.target(TargetFilters.creature()).bindEffect(destroy);
        two.target(TargetFilters.creature()).bindEffect(boost2);
        cards.put("twogroups", two);

        Card grave = card(prefix + " grave", type, subtypes);
        grave.addTrigger(trigger, ReturnCardFromGraveyardEffect.builder()
                .destination(GraveyardChoiceDestination.HAND)
                .filter(new CardTypePredicate(CardType.CREATURE))
                .targetGraveyard(true)
                .build());
        cards.put("grave", grave);

        Card modal = card(prefix + " modal", type, subtypes);
        modal.addTrigger(trigger, gainOrDraw());
        cards.put("modal", modal);

        Card modalTarget = card(prefix + " modaltarget", type, subtypes);
        modalTarget.addTrigger(trigger, new ChooseOneAtTriggerTimeEffect(new ChooseOneEffect(
                List.of(new ChooseOneEffect.ChooseOneOption("Destroy target creature",
                                new DestroyTargetPermanentEffect(), TargetFilters.creature()),
                        new ChooseOneEffect.ChooseOneOption("You gain 1 life", new GainLifeEffect(1))))));
        cards.put("modaltarget", modalTarget);
        cards.keySet().removeIf(key -> !key.equals("plain") && !key.equals(which));
        variantNames.clear();
        cards.values().forEach(c -> variantNames.add(c.getName()));
        return cards;
    }

    private void world() {
        graveCard = new GrizzlyBears();
        List<Card> graveyard = new ArrayList<>();
        graveyard.add(graveCard);
        for (int i = 0; i < 6; i++) {
            graveyard.add(new GrizzlyBears());
        }
        harness.setGraveyard(player1, graveyard);
        targetBears = addCreatureReady(player2, new GrizzlyBears());
        targetBears.tap();
    }

    private void answerAll() {
        TriggerCollectionService collection = GameTestEngineContext.get().getBean(TriggerCollectionService.class);
        for (int i = 0; i < 40; i++) {
            if (!gd.interaction.isAwaitingInput()) {
                try {
                    harness.inMutationScope(() -> collection.processNextTriggeredModalTrigger(gd));
                } catch (RuntimeException ignored) {
                }
                if (!gd.interaction.isAwaitingInput()) {
                    return;
                }
            }
            int before = gd.stack.size();
            String state = "awaiting";
            if (attempt(() -> harness.handleListChoice(player1, "Destroy target creature"), before, state)) continue;
            if (attempt(() -> harness.handleListChoice(player1, "You gain 1 life"), before, state)) continue;
            if (attempt(() -> harness.handlePermanentChosen(player1, targetBears.getId()), before, state)) continue;
            List<UUID> cards = gd.playerGraveyards.get(player1.getId()).stream().map(Card::getId).limit(1).toList();
            if (attempt(() -> harness.handleMultipleCardsChosen(player1, cards), before, state)) continue;
            if (attempt(() -> harness.handleGraveyardCardChosen(player1, 0), before, state)) continue;
            if (attempt(() -> harness.handleMultiplePermanentsChosen(player1, List.of(targetBears.getId())), before, state)) continue;
            write("    UNANSWERED " + which + " " + gd.interaction.activeInteraction().getClass().getSimpleName() + " "
                    + (gd.interaction.permanentChoiceContext() == null ? "" : gd.interaction.permanentChoiceContext().getClass().getSimpleName())
                    + " " + errors);
            return;
        }
    }

    private final List<String> errors = new ArrayList<>();

    private boolean attempt(Runnable action, int stackBefore, String stateBefore) {
        try {
            action.run();
        } catch (RuntimeException | AssertionError e) {
            errors.add(String.valueOf(e.getMessage()));
            return false;
        }
        errors.clear();
        return true;
    }

    private static void write(String line) {
        try {
            Files.writeString(OUT, line + "\n", StandardOpenOption.CREATE, StandardOpenOption.APPEND);
        } catch (Exception e) {
            throw new IllegalStateException(e);
        }
    }

    private Map<String, String> record(StackEntry entry) {
        Map<String, String> record = new LinkedHashMap<>();
        for (String name : EVENT_FIELDS) {
            try {
                Field field = StackEntry.class.getDeclaredField(name);
                field.setAccessible(true);
                record.put(name, normalize(field.get(entry), entry));
            } catch (NoSuchFieldException e) {
                record.put(name, "NOFIELD");
            } catch (IllegalAccessException e) {
                throw new IllegalStateException(e);
            }
        }
        record.put("nonTargeting", String.valueOf(entry.isNonTargeting()));
        return record;
    }

    private String normalize(Object value, StackEntry entry) {
        if (value == null) {
            return "null";
        }
        if (value instanceof Permanent permanent) {
            return "P(" + id(permanent.getId(), entry) + ")";
        }
        if (value instanceof StackEntry) {
            return "S";
        }
        if (value instanceof Card card) {
            return "C(" + (card.getId().equals(entry.getCard().getId()) ? "SELF" : card.getName()) + ")";
        }
        if (value instanceof UUID uuid) {
            return id(uuid, entry);
        }
        if (value instanceof Collection<?> collection) {
            return collection.stream().map(element -> normalize(element, entry)).toList().toString();
        }
        if (value instanceof Map<?, ?> map) {
            return "M" + map.size();
        }
        if (value instanceof String || value instanceof Number || value instanceof Boolean || value instanceof Enum<?>) {
            return value instanceof String ? "str" : value.toString();
        }
        return value.getClass().getSimpleName();
    }

    private String id(UUID uuid, StackEntry entry) {
        if (uuid.equals(entry.getSourcePermanentId())) return "SOURCE";
        if (uuid.equals(entry.getCard().getId())) return "SELFCARD";
        if (uuid.equals(player1.getId())) return "p1";
        if (uuid.equals(player2.getId())) return "p2";
        if (targetBears != null && uuid.equals(targetBears.getId())) return "targetBears";
        return "id";
    }

    private void report(String scenario) {
        answerAll();
        write("== " + scenario + " / " + which);
        Map<String, String> plain = null;
        for (String name : variantNames) {
            List<StackEntry> entries = gd.stack.stream().filter(e -> e.getCard().getName().equals(name)).toList();
            if (entries.size() != 1) {
                write("  " + name + ": " + entries.size() + " entries");
                if (entries.isEmpty()) continue;
            }
            Map<String, String> record = record(entries.getFirst());
            if (plain == null) {
                plain = record;
                continue;
            }
            List<String> diffs = new ArrayList<>();
            for (Map.Entry<String, String> field : plain.entrySet()) {
                String other = record.get(field.getKey());
                if (!field.getValue().equals(other)) {
                    diffs.add(field.getKey() + ": " + field.getValue() + " -> " + other);
                }
            }
            write("  " + name + ": " + (diffs.isEmpty() ? "SAME" : diffs.toString()));
        }
    }

        this.which = which;
        world();
        List<Integer> indices = new ArrayList<>();
        int index = 0;
        for (Card card : variants("SelfAttack", CardType.CREATURE, Triggers.attacks()).values()) {

            indices.add(index++);
        }
        harness.withAutoStop(TurnStep.DECLARE_ATTACKERS, () -> declareAttackers(player1, indices));
        report("self attack");
    }
